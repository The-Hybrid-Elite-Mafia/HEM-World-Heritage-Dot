cmake_minimum_required(VERSION 3.20)
project(WorldHeritageInstrumentCreator VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
 message(FATAL_ERROR "Windows x64 required")
endif()
add_library(core STATIC sample.cpp mmap_streamer.cpp dsp.cpp audio.cpp asio_backend.cpp)
target_compile_definitions(core PUBLIC UNICODE _UNICODE)
target_link_libraries(core PUBLIC mfplat mfreadwrite mfuuid ole32 propsys uuid winmm ksuser advapi32)
enable_language(RC)
add_executable(WorldHeritageInstrumentCreator WIN32 workstation.cpp version.rc)
target_link_libraries(WorldHeritageInstrumentCreator PRIVATE core d2d1 dwrite shell32)
if(MINGW)
 target_link_options(WorldHeritageInstrumentCreator PRIVATE -municode -static)
endif()
foreach(test portable_pack_tests workstation_tests knob_tests nonblocking_metadata_test audio_preferences_test asio_backend_test)
 add_executable(${test} ${test}.cpp)
 target_link_libraries(${test} PRIVATE core)
 if(MINGW)
  target_link_options(${test} PRIVATE -static)
  if(NOT test MATCHES "^(tests|sample_portable_test|contract_tests)$")
   target_link_options(${test} PRIVATE -municode)
  endif()
 endif()
endforeach()




# Compile test-only callback hooks in this executable, never in the app.
target_sources(asio_backend_test PRIVATE asio_backend.cpp)
target_compile_definitions(asio_backend_test PRIVATE WHIC_ASIO_BACKEND_TESTING)

if(MSVC)
 target_compile_options(core PUBLIC /arch:AVX2 /utf-8)
else()
 target_compile_options(core PUBLIC -mavx2)
endif()

add_executable(offscreen_render offscreen_render.cpp)
target_link_libraries(offscreen_render PRIVATE core d2d1 dwrite windowscodecs shell32)
if(MINGW)
 target_link_options(offscreen_render PRIVATE -municode -static)
endif()

add_executable(ui_model_tests ui_model_tests.cpp)
target_link_libraries(ui_model_tests PRIVATE core d2d1 dwrite shell32)
if(MINGW)
 target_link_options(ui_model_tests PRIVATE -municode -static)
endif()

foreach(test repair_ui_tests level_tests)
 add_executable(${test} ${test}.cpp)
 target_link_libraries(${test} PRIVATE core d2d1 dwrite shell32)
 if(MINGW)
  target_link_options(${test} PRIVATE -municode -static)
 endif()
endforeach()
